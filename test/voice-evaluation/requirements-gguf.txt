# The GGUF arm's toolchain, separate from the ONNX one because it shares almost
# nothing with it: llama.cpp for the backbone, and a neural codec for the
# waveform the backbone does not produce.
#
# PINNED, because a benchmark whose runtime moves between two runs makes the two
# figures incomparable - which is the whole point of the exercise.

# llama.cpp bindings, COMPILED FROM THE SDIST RATHER THAN A WHEEL.
#
# MEASURED 2026-09-14, and it cost a run to learn: the project's own wheel index
# at abetlen.github.io/llama-cpp-python/whl/cpu publishes files named
# `...-linux_x86_64.whl`, but the shared library inside them is MUSL-linked. On a
# glibc runner - which every `ubuntu-latest` is - importing it dies with
#
#     OSError: libc.musl-x86_64.so.1: cannot open shared object file
#
# PyPI publishes no manylinux wheel at all for this package, only the sdist. So
# there is no prebuilt path to a glibc runner and the choice is to compile. The
# build costs several minutes of a 4 vCPU runner once per cache miss, against a
# benchmark that runs for an hour; forcing musl into a glibc process to save it
# would be two libcs in one address space, which is not a saving.
llama-cpp-python==0.3.35

# Orpheus emits SNAC codes. The decoder is ~20M parameters and torch-only, so
# this arm pays for CPU torch - the beneficiary feature is "audio at all", since
# without it the backbone produces integers.
snac>=1.2.1

# NeuTTS Air's extras are NOT here. Measured 2026-09-14: `neucodec` imports
# `torchtune`, which imports `torchao`, and that chain does not resolve cleanly
# on Python 3.12 - it drags in a torchao carrying cpython-310 binaries and then
# fails on a module torchtune expects and torchao has moved. Keeping it out of
# the shared file means one model's dependency swamp cannot sink the other arm,
# which is exactly what it did: the Orpheus run that followed found no `snac`
# because the whole install had failed on NeuTTS's behalf. See
# requirements-neutts.txt.
onnxruntime==1.22.1
